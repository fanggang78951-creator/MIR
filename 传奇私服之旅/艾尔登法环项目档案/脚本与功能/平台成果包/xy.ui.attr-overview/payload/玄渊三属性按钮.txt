[@XY_UI_DRAW_REVIVE]
{{
#IF
#ACT
CloseClientBuff {revive_buff_id}

#CALL [\玄渊功能\非常驻\属性总览\玄渊三属性按钮.txt] @XY_UI_CHECK_REVIVE_ITEM
#IF
EQUAL N$XY_UI_REVIVE_EQUIP_COUNT 0
#ACT
ADDBUTTON {icon_wil_index} {revive_button_id} {revive_inactive_normal} {revive_inactive_hover} {revive_inactive_pressed} {revive_x} {revive_y} 0 -1 249#复活状态\249#未佩戴有效复活装备\249#剩余复活机会:0
BREAK
#IF
EQUAL N$XY_UI_REVIVE_OWNED 1
LARGE <$NpcRebornCount> 0
#ACT
ADDBUTTON {icon_wil_index} {revive_button_id} {revive_active_normal} {revive_active_hover} {revive_active_pressed} {revive_x} {revive_y} 0 -1 250#复活状态\251#剩余复活机会:1\254#与原生冷却共用同一窗口
BREAK
#IF
EQUAL <$REVIVALTIME> 0
EQUAL N$XY_UI_REVIVE_EQUIP_COUNT 2
#ACT
ADDBUTTON {icon_wil_index} {revive_button_id} {revive_active_normal} {revive_active_hover} {revive_active_pressed} {revive_x} {revive_y} 0 -1 250#复活状态\251#剩余复活机会:2\254#复活已就绪
BREAK
#IF
EQUAL <$REVIVALTIME> 0
EQUAL N$XY_UI_REVIVE_EQUIP_COUNT 1
#ACT
ADDBUTTON {icon_wil_index} {revive_button_id} {revive_active_normal} {revive_active_hover} {revive_active_pressed} {revive_x} {revive_y} 0 -1 250#复活状态\251#剩余复活机会:1\254#复活已就绪
BREAK
#IF
LARGE <$REVIVALTIME> 0
#ACT
MOV N$XY_UI_REVIVE_COUNTDOWN <$REVIVALTIME>
INC N$XY_UI_REVIVE_COUNTDOWN 1
SetClientBuff {icon_wil_index} {revive_buff_id} {revive_inactive_normal} <$STR(N$XY_UI_REVIVE_COUNTDOWN)> {revive_x} {revive_y} 250/复活冷却↓
BREAK
}}

[@XY_UI_AFTER_NATIVE_REVIVAL]
{{
#IF
#ACT
#CALL [\玄渊功能\非常驻\属性总览\玄渊三属性按钮.txt] @XY_UI_CHECK_REVIVE_ITEM
#IF
SMALL N$XY_UI_REVIVE_EQUIP_COUNT 2
#ACT
MOV N$XY_UI_REVIVE_OWNED 0
#CALL [\玄渊功能\非常驻\属性总览\玄渊三属性按钮.txt] @XY_UI_DRAW_REVIVE
BREAK
#IF
#ACT
MOV N$XY_UI_REVIVE_WINDOW <$REVIVALTIME>
#IF
SMALL N$XY_UI_REVIVE_WINDOW 1
#ACT
MOV N$XY_UI_REVIVE_WINDOW 60
#IF
#ACT
SetReborn 1 <$STR(N$XY_UI_REVIVE_WINDOW)>
MOV N$XY_UI_REVIVE_OWNED 1
DELAYGOTO 61 @XY_UI_REVIVE_WINDOW_END
#CALL [\玄渊功能\非常驻\属性总览\玄渊三属性按钮.txt] @XY_UI_DRAW_REVIVE
BREAK
}}
