[@XY_DONATE_TRIGGER]
{
#IF
#ACT
MOV N$XY_DONATE_ENABLE 0
MOV N$XY_DONATE_COST_COUNT 0
MOV N$XY_DONATE_COST_THRESHOLD 0
MOV N$XY_DONATE_ADD 0
MOV N$XY_DONATE_SEAL_LIMIT 0
MOV N$XY_DONATE_SEALED 0
MOV N$XY_DONATE_PERSONAL 0
MOV N$XY_DONATE_GLOBAL 0
MOV N$XY_DONATE_TOP 0
MOV N$XY_DONATE_LAST_SUCCESS 0
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 基础 功能开关 N$XY_DONATE_ENABLE
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 消耗 消耗类型 S$XY_DONATE_COST_TYPE
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 消耗 消耗名称 S$XY_DONATE_COST_NAME
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 消耗 消耗数量 N$XY_DONATE_COST_COUNT
MOV N$XY_DONATE_COST_THRESHOLD <$STR(N$XY_DONATE_COST_COUNT)>
DEC N$XY_DONATE_COST_THRESHOLD 1
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 累计 累计增加 N$XY_DONATE_ADD
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 累计 封榜阈值 N$XY_DONATE_SEAL_LIMIT
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 捐献成功 S$XY_DONATE_SUCCESS
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 条件不足 S$XY_DONATE_LACK
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 已经封榜 S$XY_DONATE_SEALED_MSG
ReadConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 封榜 N$XY_DONATE_SEALED
ReadConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 累计 N$XY_DONATE_GLOBAL
ReadConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 榜一金额 N$XY_DONATE_TOP
ReadConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 榜一玩家 S$XY_DONATE_TOP_NAME
ReadConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 玩家 <$USERNAME> N$XY_DONATE_PERSONAL

#IF
EQUAL N$XY_DONATE_ENABLE 0
#ACT
BREAK

#IF
EQUAL N$XY_DONATE_SEALED 1
#ACT
MESSAGEBOX <$STR(S$XY_DONATE_SEALED_MSG)>
BREAK

#IF
EQUAL S$XY_DONATE_COST_TYPE 原生灵符
CHECKGAMEGIRD > <$STR(N$XY_DONATE_COST_THRESHOLD)>
#ACT
GAMEGIRD - <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
EQUAL S$XY_DONATE_COST_TYPE 原生金币
CHECKGOLD <$STR(N$XY_DONATE_COST_COUNT)>
#ACT
GOLDCOUNT - <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
EQUAL S$XY_DONATE_COST_TYPE 原生元宝
CHECKGAMEGOLD > <$STR(N$XY_DONATE_COST_THRESHOLD)>
#ACT
GAMEGOLD - <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
EQUAL S$XY_DONATE_COST_TYPE 原生金刚石
CHECKGAMEDIAMOND <$STR(N$XY_DONATE_COST_COUNT)>
#ACT
GAMEDIAMOND - <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
EQUAL S$XY_DONATE_COST_TYPE 材料
EQUAL S$XY_DONATE_COST_NAME 灵符
CHECKGAMEGIRD > <$STR(N$XY_DONATE_COST_THRESHOLD)>
#ACT
GAMEGIRD - <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
EQUAL S$XY_DONATE_COST_TYPE 账户金刚石
CHECKGAMEDIAMOND <$STR(N$XY_DONATE_COST_COUNT)>
#ACT
GAMEDIAMOND - <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
EQUAL S$XY_DONATE_COST_TYPE 材料
NOT EQUAL S$XY_DONATE_COST_NAME 灵符
CHECKITEM <$STR(S$XY_DONATE_COST_NAME)> <$STR(N$XY_DONATE_COST_COUNT)>
#ACT
TAKE <$STR(S$XY_DONATE_COST_NAME)> <$STR(N$XY_DONATE_COST_COUNT)>
GOTO @XY_DONATE_APPLY

#IF
#ACT
MESSAGEBOX <$STR(S$XY_DONATE_LACK)>
BREAK

[@XY_DONATE_APPLY]
#IF
#ACT
MOV N$XY_DONATE_LAST_SUCCESS 1
INC N$XY_DONATE_PERSONAL <$STR(N$XY_DONATE_ADD)>
INC N$XY_DONATE_GLOBAL <$STR(N$XY_DONATE_ADD)>
WriteConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 玩家 <$USERNAME> <$STR(N$XY_DONATE_PERSONAL)>
WriteConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 累计 <$STR(N$XY_DONATE_GLOBAL)>

#IF
LARGE N$XY_DONATE_PERSONAL <$STR(N$XY_DONATE_TOP)>
#ACT
MOV N$XY_DONATE_TOP <$STR(N$XY_DONATE_PERSONAL)>
MOV S$XY_DONATE_TOP_NAME <$USERNAME>
WriteConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 榜一金额 <$STR(N$XY_DONATE_TOP)>
WriteConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 榜一玩家 <$STR(S$XY_DONATE_TOP_NAME)>

#IF
LARGE N$XY_DONATE_SEAL_LIMIT 0
LARGE N$XY_DONATE_GLOBAL <$STR(N$XY_DONATE_SEAL_LIMIT)>
#ACT
MOV N$XY_DONATE_SEALED 1
WriteConfigFileItem ..\QuestDiary\玄渊数据\xy_donate\捐献状态.txt 全服 封榜 1

#IF
#ACT
SENDMSG 6 <$STR(S$XY_DONATE_SUCCESS)>
BREAK
}
