[@XY_DONATE_TRIGGER]
{
#IF
#ACT
MOV N$XY_DONATE_ENABLE 0
MOV N$XY_DONATE_COST_COUNT 0
MOV N$XY_DONATE_COST_THRESHOLD 0
MOV N$XY_DONATE_LAST_SUCCESS 0
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 基础 功能开关 N$XY_DONATE_ENABLE
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 消耗 消耗类型 S$XY_DONATE_COST_TYPE
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 消耗 消耗数量 N$XY_DONATE_COST_COUNT
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 激活成功 S$XY_DONATE_SUCCESS
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 条件不足 S$XY_DONATE_LACK
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 已经激活 S$XY_DONATE_ALREADY
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 授予失败 S$XY_DONATE_GRANT_FAILED
ReadConfigFileItem ..\QuestDiary\玄渊配置\沙城捐献配置.txt 文案 功能关闭 S$XY_DONATE_DISABLED
MOV N$XY_DONATE_COST_THRESHOLD <$STR(N$XY_DONATE_COST_COUNT)>
DEC N$XY_DONATE_COST_THRESHOLD 1

#IF
EQUAL N$XY_DONATE_ENABLE 0
#ACT
MESSAGEBOX <$STR(S$XY_DONATE_DISABLED)>
BREAK

#IF
CHECKFENGHAO 沙城捐献
#ACT
SetNewFengHaoValue 沙城捐献 0 = 10
MESSAGEBOX <$STR(S$XY_DONATE_ALREADY)>
BREAK

#IF
NOT EQUAL S$XY_DONATE_COST_TYPE 原生灵符
#ACT
MESSAGEBOX <$STR(S$XY_DONATE_DISABLED)>
BREAK

#IF
NOT EQUAL N$XY_DONATE_COST_COUNT 300
#ACT
MESSAGEBOX <$STR(S$XY_DONATE_DISABLED)>
BREAK

#IF
CHECKGAMEGIRD > <$STR(N$XY_DONATE_COST_THRESHOLD)>
#ACT
GIVEFENGHAO 沙城捐献 1
GOTO @XY_DONATE_FINISH_LINGFU

#IF
#ACT
MESSAGEBOX <$STR(S$XY_DONATE_LACK)>
BREAK

[@XY_DONATE_FINISH_LINGFU]
#IF
CHECKFENGHAO 沙城捐献
#ACT
GAMEGIRD - <$STR(N$XY_DONATE_COST_COUNT)>
SetNewFengHaoValue 沙城捐献 0 = 10
MOV N$XY_DONATE_LAST_SUCCESS 1
SENDMSG 6 <$STR(S$XY_DONATE_SUCCESS)>
BREAK
#ELSEACT
MESSAGEBOX <$STR(S$XY_DONATE_GRANT_FAILED)>
BREAK
}
