[@XY_RAGE_TRIGGER]
{
#IF
#ACT
MOV N$XY_RAGE_ENABLE 0
MOV N$XY_RAGE_COST_COUNT 0
MOV N$XY_RAGE_COST_THRESHOLD 0
MOV N$XY_RAGE_POWER 100
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 基础 功能开关 N$XY_RAGE_ENABLE
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 消耗 消耗类型 S$XY_RAGE_COST_TYPE
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 消耗 消耗名称 S$XY_RAGE_COST_NAME
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 消耗 消耗数量 N$XY_RAGE_COST_COUNT
MOV N$XY_RAGE_COST_THRESHOLD <$STR(N$XY_RAGE_COST_COUNT)>
DEC N$XY_RAGE_COST_THRESHOLD 1
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 属性 神力倍攻 N$XY_RAGE_POWER
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 文案 开启成功 S$XY_RAGE_SUCCESS
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 文案 条件不足 S$XY_RAGE_LACK
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 文案 重复开启 S$XY_RAGE_REPEAT

#IF
EQUAL N$XY_RAGE_ENABLE 0
#ACT
BREAK

#IF
EQUAL N$XY_RAGE_ACTIVE 1
#ACT
MESSAGEBOX <$STR(S$XY_RAGE_REPEAT)>
BREAK

#IF
EQUAL S$XY_RAGE_COST_TYPE 原生灵符
CHECKGAMEGIRD > <$STR(N$XY_RAGE_COST_THRESHOLD)>
#ACT
GAMEGIRD - <$STR(N$XY_RAGE_COST_COUNT)>
GOTO @XY_RAGE_APPLY

#IF
EQUAL S$XY_RAGE_COST_TYPE 金币
CHECKGOLD <$STR(N$XY_RAGE_COST_COUNT)>
#ACT
TAKE 金币 <$STR(N$XY_RAGE_COST_COUNT)>
GOTO @XY_RAGE_APPLY

#IF
EQUAL S$XY_RAGE_COST_TYPE 材料
CHECKITEM <$STR(S$XY_RAGE_COST_NAME)> <$STR(N$XY_RAGE_COST_COUNT)>
#ACT
TAKE <$STR(S$XY_RAGE_COST_NAME)> <$STR(N$XY_RAGE_COST_COUNT)>
GOTO @XY_RAGE_APPLY

#IF
#ACT
MESSAGEBOX <$STR(S$XY_RAGE_LACK)>
BREAK

[@XY_RAGE_APPLY]
#IF
#ACT
MOV N$XY_RAGE_ACTIVE 1
POWERRATE <$STR(N$XY_RAGE_POWER)> 0 0 1 0
SENDMSG 6 <$STR(S$XY_RAGE_SUCCESS)>
BREAK
}

[@XY_RAGE_REFRESH]
{
#IF
EQUAL N$XY_RAGE_ACTIVE 1
#ACT
MOV N$XY_RAGE_POWER 100
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 属性 神力倍攻 N$XY_RAGE_POWER
POWERRATE <$STR(N$XY_RAGE_POWER)> 0 0 1 0
BREAK
}

[@XY_RAGE_DIE]
{
#IF
EQUAL N$XY_RAGE_ACTIVE 0
#ACT
BREAK

#IF
#ACT
MOV N$XY_RAGE_CLEAR 0
MOV N$XY_RAGE_REWARD_COUNT 0
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 生命周期 死亡清除 N$XY_RAGE_CLEAR
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 奖励 击杀奖励类型 S$XY_RAGE_REWARD_TYPE
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 奖励 击杀奖励名称 S$XY_RAGE_REWARD_NAME
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 奖励 击杀奖励数量 N$XY_RAGE_REWARD_COUNT

#IF
EQUAL N$XY_RAGE_CLEAR 1
EQUAL S$XY_RAGE_REWARD_TYPE 账户金刚石
LARGE N$XY_RAGE_REWARD_COUNT 0
KillByHum
#ACT
<$KILLER>.GAMEDIAMOND + <$STR(N$XY_RAGE_REWARD_COUNT)>

#IF
EQUAL N$XY_RAGE_CLEAR 1
#ACT
MOV N$XY_RAGE_ACTIVE 0
MOV N$XY_RAGE_CLEAR 0
MOV N$XY_RAGE_REWARD_COUNT 0
POWERRATE 100 0 0 1 0
BREAK

#IF
#ACT
BREAK
}

[@XY_RAGE_OFFLINE]
{
#IF
EQUAL N$XY_RAGE_ACTIVE 0
#ACT
BREAK

#IF
#ACT
MOV N$XY_RAGE_CLEAR 0
ReadConfigFileItem ..\QuestDiary\玄渊配置\狂暴配置.txt 生命周期 下线清除 N$XY_RAGE_CLEAR

#IF
EQUAL N$XY_RAGE_CLEAR 1
#ACT
MOV N$XY_RAGE_ACTIVE 0
MOV N$XY_RAGE_CLEAR 0
POWERRATE 100 0 0 1 0
BREAK

#IF
#ACT
BREAK
}
